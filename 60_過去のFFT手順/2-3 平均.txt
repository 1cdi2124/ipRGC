Sub CopyCalcAndGraph_OnAverageSheet()

    Dim wsAvg As Worksheet
    Dim wsSrc As Worksheet
    Dim lastRow As Long
    Dim col As Long
    Dim sheetName As String
    Dim r As Long
    Dim cht As ChartObject
    Dim startLeft As Double
    Dim startTop As Double
    Dim graphWidth As Double
    Dim graphHeight As Double

    ' 平均シート
    Set wsAvg = ThisWorkbook.Worksheets("平均")

    ' =================================================
    ' ① G列（100-0） → 平均シート C～G
    ' =================================================
    For col = 3 To 7
        sheetName = wsAvg.Cells(1, col).Value
        Set wsSrc = ThisWorkbook.Worksheets(sheetName)

        lastRow = wsSrc.Cells(wsSrc.Rows.Count, 7).End(xlUp).Row

        wsAvg.Cells(2, col).Resize(lastRow - 1, 1).Value = _
            wsSrc.Cells(2, 7).Resize(lastRow - 1, 1).Value
    Next col

    ' =================================================
    ' ② H列（200-0） → 平均シート K～O
    ' =================================================
    For col = 11 To 15
        sheetName = wsAvg.Cells(1, col).Value
        Set wsSrc = ThisWorkbook.Worksheets(sheetName)

        lastRow = wsSrc.Cells(wsSrc.Rows.Count, 8).End(xlUp).Row

        wsAvg.Cells(2, col).Resize(lastRow - 1, 1).Value = _
            wsSrc.Cells(2, 8).Resize(lastRow - 1, 1).Value
    Next col

    ' =================================================
    ' ③ C～G の行平均 → H列
    ' =================================================
    lastRow = wsAvg.Cells(wsAvg.Rows.Count, 3).End(xlUp).Row

    For r = 2 To lastRow
        wsAvg.Cells(r, 8).Value = Application.WorksheetFunction.Average( _
            wsAvg.Range(wsAvg.Cells(r, 3), wsAvg.Cells(r, 7)) _
        )
    Next r

    ' =================================================
    ' ④ K～O の行平均 → P列
    ' =================================================
    For r = 2 To lastRow
        wsAvg.Cells(r, 16).Value = Application.WorksheetFunction.Average( _
            wsAvg.Range(wsAvg.Cells(r, 11), wsAvg.Cells(r, 15)) _
        )
    Next r

    ' =================================================
    ' ⑤ 既存グラフ削除（平均シート）
    ' =================================================
    For Each cht In wsAvg.ChartObjects
        cht.Delete
    Next cht

    ' =================================================
    ' ⑥ 平均グラフ作成（H列・P列）
    ' =================================================
    startLeft = wsAvg.Range("R2").Left
    startTop = wsAvg.Range("R2").Top
    graphWidth = 360
    graphHeight = 220

    ' ---- H列：100-0 平均 ----
    Set cht = wsAvg.ChartObjects.Add(startLeft, startTop, graphWidth, graphHeight)
    With cht.Chart
        .ChartType = xlLine
        .SeriesCollection.NewSeries
        .SeriesCollection(1).Values = wsAvg.Range("H2:H" & lastRow)
        .SeriesCollection(1).Name = "100-0 平均"

        With .Axes(xlValue)
            .MinimumScale = -0.05
            .MaximumScale = 0.05
        End With

        .HasTitle = True
        .ChartTitle.Text = "100-0 平均"
    End With

    ' ---- P列：200-0 平均 ----
    Set cht = wsAvg.ChartObjects.Add(startLeft, startTop + graphHeight, graphWidth, graphHeight)
    With cht.Chart
        .ChartType = xlLine
        .SeriesCollection.NewSeries
        .SeriesCollection(1).Values = wsAvg.Range("P2:P" & lastRow)
        .SeriesCollection(1).Name = "200-0 平均"

        With .Axes(xlValue)
            .MinimumScale = -0.05
            .MaximumScale = 0.05
        End With

        .HasTitle = True
        .ChartTitle.Text = "200-0 平均"
    End With

End Sub
